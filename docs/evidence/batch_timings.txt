#!/usr/bin/env bash
# Time each job of the one-time batch build (README section (c), steps 1-6 plus gold_br7),
# in README order, one after another, and write docs/evidence/batch_timings.txt.
# Run with Postgres and Cassandra up and nothing else running (no streaming job, no API,
# no web), so no job competes for CPU:
#     bash batch_timings.sh /home/dat/shop-insight
# stdout of each job goes into the evidence file; stderr (Spark's own log) stays in
# $LOGS/<job>.err next to this script.
set -uo pipefail
SRC=$(realpath "$0")
LOGS="$(dirname "$SRC")/batch_timings_logs"
cd "$1"

OUT=docs/evidence/batch_timings.txt
mkdir -p "$LOGS"
rm -f "$LOGS"/*.out "$LOGS"/*.err    # a stale log from an earlier run must not reach the evidence file

JOBS=(
  ingestion/load_bronze.py
  processing/batch_silver.py
  ingestion/gen_catalog.py
  ingestion/gen_inventory.py
  processing/batch_warehouse.py
  processing/batch_gold.py
  processing/gold_br7.py
)

rule() { printf '%s\n' "$(printf '=%.0s' {1..100})" "$1" "$(printf '=%.0s' {1..100})"; }

started_at=$(date --iso-8601=seconds)
summary=()
total=0
failed=""

for i in "${!JOBS[@]}"; do
  job=${JOBS[$i]}
  name=$(basename "$job" .py)
  t0=$(date +%s.%N)
  .venv/bin/python "$job" > "$LOGS/$name.out" 2> "$LOGS/$name.err"
  rc=$?
  t1=$(date +%s.%N)
  secs=$(echo "$t1 - $t0" | bc)
  total=$(echo "$total + $secs" | bc)
  last=$(grep -E 'OK|MISMATCH|FAILED' "$LOGS/$name.out" | tail -1)
  summary+=("$(printf '%-4s %-22s %9.1f %5s   %s' "$((i + 1))" "$name.py" "$secs" "$rc" "$last")")
  echo "$(date +%T) $name.py rc=$rc ${secs}s $last"
  if [ "$rc" -ne 0 ]; then failed=$name; break; fi
done

{
  echo "batch_timings - wall-clock time of each one-time batch job, load_bronze -> gold_br7 (sprint task T02)"
  echo
  rule "HEADER"
  echo "started     $started_at"
  echo "finished    $(date --iso-8601=seconds)"
  echo "machine     $(hostname) | $(uname -sr) | $(nproc) logical cores | $(free -g | awk '/Mem:/ {print $2}') GB RAM"
  echo "versions    Python $(.venv/bin/python -c 'import platform; print(platform.python_version())') | PySpark $(.venv/bin/python -c 'import pyspark; print(pyspark.__version__)') | $(java -version 2>&1 | head -1)"
  echo "spark conf  $(grep -v '^#' conf/spark-defaults.conf | grep . | tr '\n' ';' | sed 's/;$//; s/;/; /g') (conf/spark-defaults.conf)"
  echo "docker      $(docker ps --format '{{.Names}} {{.Image}} {{.Status}}' | tr '\n' ';' | sed 's/;$//; s/;/; /g')"
  echo "repo        $(pwd) @ $(git rev-parse --short HEAD) ($(git rev-parse --abbrev-ref HEAD))"
  echo "input       data/raw/2019-Nov.csv, $(du -h data/raw/2019-Nov.csv | cut -f1)"
  echo "command     bash batch_timings.sh /home/dat/shop-insight (source at the end of this file)"
  echo
  echo "Jobs run one after another, nothing else running. Wall clock includes JVM start and stop,"
  echo "so it is a few seconds above the 'took' line some jobs print themselves."
  echo
  rule "1  WALL-CLOCK TIME PER JOB"
  printf '%-4s %-22s %9s %5s   %s\n' "step" "job" "seconds" "exit" "last status line"
  printf '%s\n' "${summary[@]}"
  printf '%-4s %-22s %9.1f\n' "" "total" "$total"
  if [ -n "$failed" ]; then echo; echo "STOPPED: $failed.py exited non-zero; later jobs were not run."; fi
  echo
  rule "2  STDOUT OF EACH JOB"
  for job in "${JOBS[@]}"; do
    name=$(basename "$job" .py)
    [ -f "$LOGS/$name.out" ] || continue
    echo "---- $job"
    cat "$LOGS/$name.out"
    if [ "$name" = "$failed" ]; then echo "---- last 40 lines of stderr"; tail -40 "$LOGS/$name.err"; fi
    echo
  done
  rule "SOURCE"
  cat "$SRC"
} > "$OUT"

echo "wrote $OUT"
[ -z "$failed" ]
