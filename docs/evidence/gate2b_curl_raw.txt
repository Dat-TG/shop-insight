== Source: time_screens.sh ==
#!/usr/bin/env bash
# T14 step 2: time each of the 8 screens' endpoints. 3 warm-up calls (not kept), then 20 timed calls.
# Per call: server-side meta.ms (QueryMeta) and client-side curl time_total (HTTP round trip incl. JSON).
set -euo pipefail
BASE=${BASE:-http://localhost:8000}
declare -a EP=(
  "BR1 /api/br1/realtime?event_date=2019-11-17&category_l1=all"
  "BR2 /api/br2/funnel?start_date=2019-11-01&end_date=2019-11-30&category_l1=all"
  "BR3 /api/br3/trending"
  "BR4 /api/br4/segments?snapshot_date=2019-11-30&segment=all"
  "BR5 /api/br5/carts?event_date=2019-11-17"
  "BR6 /api/br6/stockout"
  "BR7 /api/br7/affinity?product_id=1005115"
  "BR8 /api/br8/alerts?event_date=2019-11-17&severity=all"
)
echo -e "br\trun\tmeta_ms\tcurl_total_ms\twarehouse\tpartitions\trows\thttp"
for e in "${EP[@]}"; do
  br=${e%% *}; path=${e#* }
  for i in $(seq -3 20); do
    out=$(curl -s -o body.json -w '%{http_code} %{time_total}' "$BASE$path")
    code=${out% *}; tt=${out#* }
    [[ $i -le 0 ]] && continue
    read -r ms wh parts rows < <(python3 -c 'import json,sys; m=json.load(open(sys.argv[1]))["meta"]; print(m["ms"], m["warehouse"], m["partitions"], m["rows"])' body.json)
    printf '%s\t%d\t%s\t%.1f\t%s\t%s\t%s\t%s\n' "$br" "$i" "$ms" "$(python3 -c "print(float('$tt')*1000)")" "$wh" "$parts" "$rows" "$code"
  done
done
