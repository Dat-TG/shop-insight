Why INT96 has no min/max: parquet-format leaves the sort order of INT96 undefined
(parquet.thrift ColumnOrder: 'INT96 (only used for legacy timestamps) - undefined'), so
parquet-mr writes no min_value/max_value for it, and Spark pushes timestamp predicates only
for TIMESTAMP_MICROS / TIMESTAMP_MILLIS. Fix: spark.sql.parquet.outputTimestampType=TIMESTAMP_MICROS.

"""E1 for the Group 15 indexing presentation (plan pages 11-12, Canva pages 12-13).

Read-only on data/. One Spark session, same conf as processing/kata.py
(SPARK_CONF_DIR=conf/ before the JVM starts), plus DuckDB for the Parquet footers.
Prints the evidence text to stdout.

Run from anywhere:
    /home/dat/shop-insight/.venv/bin/python presentation_parquet_run.py > presentation_parquet_run.txt
"""

import os

from pathlib import Path

REPO = Path("/home/dat/shop-insight")
os.environ["SPARK_CONF_DIR"] = str(REPO / "conf")

from pyspark.sql import SparkSession, functions as F  # noqa: E402

SILVER = REPO / "data" / "silver"
DIM_DATE = REPO / "data" / "warehouse" / "dim_date"
FACT_EVENT = REPO / "data" / "warehouse" / "fact_event"
RULE = "=" * 100


def section(title):
    print()
    print(RULE)
    print(title)
    print(RULE)


def explain(df):
    buf = io.StringIO()
    with contextlib.redirect_stdout(buf):
        df.explain(mode="formatted")
    return buf.getvalue().rstrip()


def last_scan_metrics(spark):
    """Scan-node metrics of the most recent SQL execution, from the Spark UI REST API."""
    sc = spark.sparkContext
    url = f"{sc.uiWebUrl}/api/v1/applications/{sc.applicationId}/sql?details=true&planDescription=false&length=1000"
    with urllib.request.urlopen(url) as r:
        executions = json.load(r)
    last = max(executions, key=lambda e: e["id"])
    out = []
    for node in last["nodes"]:
        if node["nodeName"].startswith("Scan parquet"):
            m = {x["name"]: x["value"] for x in node["metrics"]}
            keep = ["number of files read", "number of partitions read", "size of files read",
                    "number of output rows", "static number of files read", "dynamic partition pruning time"]
            out.append((node["nodeName"], {k: m[k] for k in keep if k in m}))
    return out


def run(spark, label, df):
    section(label)
    print(explain(df))
    rows = df.collect()
    print()
    print(f"RESULT  {[r.asDict() for r in rows]}")
    for name, m in last_scan_metrics(spark):
        print(f"METRICS {name}: {m}")

          f"(+07) by processing/batch_silver.py")

    files = sorted(SILVER.glob("event_date=*/event_type=*/*.parquet"))
    dates = sorted({p.parent.name for p in dirs})
    print(f"            silver = {len(dates)} event_date directories ({dates[0]} .. {dates[-1]}), "

    print("command     .venv/bin/python presentation_parquet_run.py (source at the end of this file)")
    print()

    print("    (spark.sql.parquet.recordLevelFilter.enabled=false), so output rows = all rows means no")
    print("    row group was skipped.")
    print("  - PushedFilters prints timestamp literals in the JVM default time zone (+07 on this machine),")
    print("    while the Filter node prints them in the session time zone (UTC). Same instant, different")
    print("    text: crop the PushedFilters line out of any overlay.")

    section("1. EFFECTIVE DEFAULTS (spark.conf.get; conf/spark-defaults.conf sets none of these)")
    for key in ["spark.sql.parquet.outputTimestampType",
                "spark.sql.parquet.filterPushdown",
                "spark.sql.parquet.recordLevelFilter.enabled",
                "spark.sql.optimizer.dynamicPartitionPruning.enabled",
                "spark.sql.optimizer.dynamicPartitionPruning.reuseBroadcastOnly",
                "spark.sql.autoBroadcastJoinThreshold",
                "spark.sql.join.preferSortMergeJoin",
                "spark.sql.streaming.stateStore.providerClass",
                "spark.sql.session.timeZone"]:
        print(f"{key} = {spark.conf.get(key)}")

    silver = spark.read.parquet(str(SILVER))
    day = dt.date(2019, 11, 15)

    run(spark, "2. Q0  no filter: every directory is read",
        silver.agg(F.count(F.lit(1)).alias("rows")))

    run(spark, "3. Q1  one day and one event type, both partition columns: 1 directory, 1 file",
        silver.where((F.col("event_date") == F.lit(day)) & (F.col("event_type") == "purchase"))
              .agg(F.count(F.lit(1)).alias("rows"), F.sum("price").alias("gmv")))

    run(spark, "4. Q2  one hour on event_time only (not a partition column; INT96, no min/max)",
        silver.where((F.col("event_time") >= F.lit("2019-11-15 10:00:00").cast("timestamp"))
                     & (F.col("event_time") < F.lit("2019-11-15 11:00:00").cast("timestamp")))
              .agg(F.count(F.lit(1)).alias("rows")))

    dim_date = spark.read.parquet(str(DIM_DATE))
    print()
    print(f"dim_date columns: {dim_date.columns}")
    picked = dim_date.where((F.col("month") == 11) & (F.col("day_of_month") <= 3)).select("full_date")
    run(spark, "5. Q3  dynamic partition pruning: join on the partition column, filter only on the small side",
        silver.join(picked, silver["event_date"] == picked["full_date"])
              .agg(F.count(F.lit(1)).alias("rows")))

    spark.stop()

    section("6. PARQUET FOOTERS (DuckDB, no Spark)")
    import duckdb
    con = duckdb.connect()
    print(f"duckdb {duckdb.__version__}")
    for name, glob in [("silver", f"{SILVER}/*/*/*.parquet"), ("fact_event", f"{FACT_EVENT}/*/*.parquet")]:
        print()
        print(f"-- {name}: per-column statistics, every file")
        q = f"""
            select path_in_schema as col, type, count(*) as chunks,
                   count(stats_min_value) as with_min, count(stats_max_value) as with_max
            from parquet_metadata('{glob}')
            where path_in_schema in ('event_time', 'product_id', 'price')
            group by all order by col"""
        for r in con.sql(q).fetchall():
            print(f"   {r[0]:<11} type={r[1]:<6} chunks={r[2]:<4} with_min={r[3]:<4} with_max={r[4]}")
        q = f"""
            select count(*) as files, min(num_row_groups), max(num_row_groups),
                   sum(num_rows), min(created_by), max(created_by)
            from parquet_file_metadata('{glob}')"""
        f, rg_min, rg_max, rows, cb_min, cb_max = con.sql(q).fetchone()
        print(f"   files={f} row_groups_per_file min={rg_min} max={rg_max} rows={rows:,} created_by={cb_min!r}"
              + ("" if cb_min == cb_max else f" .. {cb_max!r}"))
    print()
    print("Why INT96 has no min/max: parquet-format leaves the sort order of INT96 undefined")
    print("(parquet.thrift ColumnOrder: 'INT96 (only used for legacy timestamps) - undefined'), so")
    print("parquet-mr writes no min_value/max_value for it, and Spark pushes timestamp predicates only")
    print("for TIMESTAMP_MICROS / TIMESTAMP_MILLIS. Fix: spark.sql.parquet.outputTimestampType=TIMESTAMP_MICROS.")

    section("SOURCE OF THIS RUN (presentation_parquet_run.py)")
    print(Path(__file__).read_text().rstrip())
