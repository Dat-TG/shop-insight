How to read the table
- events     rows the job read after clean()'s row rules. BR3 reads view and cart only, BR8 purchases only.
- events/s   events / (time from the job's first to last micro-batch + one 5 s trigger). This is the sustained
             rate during the replay. At --speed 60 it equals what the replayer offers (one file every 5 s), so
             it measures the source, not Spark's ceiling.
- trig p50/p95/max   durationMs.triggerExecution per micro-batch in ms, nearest rank: the time Spark took to
             process one micro-batch end to end, Cassandra writes included. Batch #0 carries JVM and query warm-up.
- proc/s p50 events / triggerExecution per batch: how fast a batch is chewed through while it runs, i.e. the
             headroom above the offered rate. It is not a measured maximum; no overload test was run.
- An event reaches Cassandra at most one trigger interval (5 s) plus one triggerExecution after its file lands.

"""Streaming throughput and micro-batch latency from one ./run_demo.sh stream log (sprint task T14 step 6)."""
import math, re, sys, datetime as dt
L, R = sys.argv[1], sys.argv[2]
P = re.compile(r'^(\S+ \S+) \[(\w+)\] batch (\d+): ([\d,]+) events after cleaning, watermark (.*?), ([\d,]+) ms')
ts = lambda s: dt.datetime.strptime(s, '%Y-%m-%d %H:%M:%S.%f')
def pr(v, p):
    v = sorted(v); return v[math.ceil(p * len(v) / 100) - 1]
jobs = {}
for line in open(L):
    m = P.match(line)
    if m:
        jobs.setdefault(m[2], []).append((ts(m[1]), int(m[3]), int(m[4].replace(',', '')), int(m[6].replace(',', ''))))
files = [l for l in open(R) if re.search(r' file \d{6} ', l)]
rows = sum(int(re.search(r'\s([\d,]+) rows', l)[1].replace(',', '')) for l in files)
f0, f1 = ts(files[0][:23]), ts(files[-1][:23])
print(f'stream log    {L}')
print(f'replayer log  {R}')
print(f'replayer      {len(files)} files, {rows:,} rows, first file {files[0][:23]}, last {files[-1][:23]} '
      f'({(f1 - f0).total_seconds():.1f} s); offered rate {rows / ((f1 - f0).total_seconds() + 5):.1f} rows/s '
      f'(rows / (span + one 5 s file interval))')
print()
print(f"{'job':28}{'batches':>8}{'events':>9}{'events/s':>10}{'trig p50':>10}{'p95':>7}{'max':>7}"
      f"{'max at':>8}{'proc/s p50':>12}")
for j, r in jobs.items():
    ev = sum(x[2] for x in r)
    span = (r[-1][0] - r[0][0]).total_seconds() + 5          # n batches cover n trigger intervals of 5 s
    d = [x[3] for x in r]
    mx = max(r, key=lambda x: x[3])
    proc = [x[2] / (x[3] / 1000) for x in r if x[3] > 0]
    print(f'{j:28}{len(r):>8}{ev:>9,}{ev / span:>10.1f}{pr(d, 50):>10,}{pr(d, 95):>7,}{mx[3]:>7,}{"#" + str(mx[1]):>8}'
          f'{pr(proc, 50):>12,.0f}')
